fetch(`${SCRIPT_URL}?action=getCatalog`) .then(res => res.json()) .then(data => { const catalogDiv = document.getElementById('catalog'); catalogDiv.innerHTML = ''; if (data.status === 'success' && data.products && data.products.length > 0) { data.products.forEach((prod) => { const card = document.createElement('div'); card.className = 'card'; card.innerHTML = `
${prod.Item_Name}
${prod.Selling_Price} ETB
`; catalogDiv.appendChild(card); }); } else { catalogDiv.innerHTML = '
No stock items found in Store_Stock.
'; } }) .catch(err => { document.getElementById('catalog').innerHTML = '
Failed to connect to store.
'; });
